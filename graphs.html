Program 1: Enter Vertices and Edges and Represent in Adjacency Matrix

#include <stdio.h>

int main()
{
    int n, e, i, u, v;
    int graph[20][20] = {0};

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++)
    {
        printf("Enter edge (u v): ");
        scanf("%d %d", &u, &v);

        graph[u][v] = 1;
        graph[v][u] = 1;
    }

    printf("\nAdjacency Matrix:\n");

    for (i = 0; i < n; i++)
    {
        for (int j = 0; j < n; j++)
        {
            printf("%d ", graph[i][j]);
        }

        printf("\n");
    }

    return 0;
}

Program 2: Create Graph as an Adjacency List with Insertion and Deletion

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int vertex;
    struct Node *next;
};

struct Node *adj[20];

void insertEdge(int u, int v)
{
    struct Node *newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->vertex = v;
    newNode->next = adj[u];
    adj[u] = newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->vertex = u;
    newNode->next = adj[v];
    adj[v] = newNode;
}

void deleteOne(int u, int v)
{
    struct Node *temp = adj[u];
    struct Node *prev = NULL;

    while (temp != NULL && temp->vertex != v)
    {
        prev = temp;
        temp = temp->next;
    }

    if (temp != NULL)
    {
        if (prev == NULL)
        {
            adj[u] = temp->next;
        }
        else
        {
            prev->next = temp->next;
        }

        free(temp);
    }
}

void deleteEdge(int u, int v)
{
    deleteOne(u, v);
    deleteOne(v, u);
}

void display(int n)
{
    int i;
    struct Node *temp;

    printf("\nAdjacency List:\n");

    for (i = 0; i < n; i++)
    {
        printf("%d: ", i);
        temp = adj[i];

        while (temp != NULL)
        {
            printf("%d -> ", temp->vertex);
            temp = temp->next;
        }

        printf("NULL\n");
    }
}

int main()
{
    int n, e, i, u, v, choice;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter number of initial edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++)
    {
        printf("Enter edge (u v): ");
        scanf("%d %d", &u, &v);
        insertEdge(u, v);
    }

    do
    {
        printf("\n1. Insert Edge\n");
        printf("2. Delete Edge\n");
        printf("3. Display\n");
        printf("0. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &choice);

        if (choice == 1)
        {
            printf("Enter edge (u v): ");
            scanf("%d %d", &u, &v);
            insertEdge(u, v);
        }
        else if (choice == 2)
        {
            printf("Enter edge to delete (u v): ");
            scanf("%d %d", &u, &v);
            deleteEdge(u, v);
        }
        else if (choice == 3)
        {
            display(n);
        }

    } while (choice != 0);

    return 0;
}

Program 3: Traverse Adjacency Matrix with Depth First Search (DFS)

#include <stdio.h>

int graph[20][20], visited[20], n;

void DFS(int vertex)
{
    int i;

    printf("%d ", vertex);
    visited[vertex] = 1;

    for (i = 0; i < n; i++)
    {
        if (graph[vertex][i] == 1 && visited[i] == 0)
        {
            DFS(i);
        }
    }
}

Program 5: Traverse Adjacency Matrix with Breadth First Search (BFS)

#include <stdio.h>

int graph[20][20], visited[20], queue[20];
int n, front = 0, rear = -1;

void BFS(int start)
{
    int i, vertex;

    visited[start] = 1;
    queue[++rear] = start;

    while (front <= rear)
    {
        vertex = queue[front++];
        printf("%d ", vertex);

        for (i = 0; i < n; i++)
        {
            if (graph[vertex][i] == 1 && visited[i] == 0)
            {
                visited[i] = 1;
                queue[++rear] = i;
            }
        }
    }
}

        graph[u][v] = 1;
        graph[v][u] = 1;

    printf("Enter starting vertex: ");

#include <stdlib.h>

struct Node
{
    int vertex;
    struct Node *next;

struct Node *adj[20];

{
    struct Node *newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->vertex = v;

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->vertex = u;
    newNode->next = adj[v];

}

void BFS(int start)
{
    int front = 0, rear = 0;
    int vertex;
    struct Node *temp;

    visited[start] = 1;
    queue[rear++] = start;

    while (front < rear)
    {
