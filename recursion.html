Program 1: Calculate Factorial of a Number Using Recursion

#include <stdio.h>

int factorial(int n)
{
    if (n == 0 || n == 1)
    {
        return 1;
    }

    return n * factorial(n - 1);
}

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    printf("Factorial = %d", factorial(n));

    return 0;
}
Input: 
Enter a number: 5
Output: 
Factorial = 120

Program 2: Calculate Fibonacci of a Number Using Recursion

#include <stdio.h>

int fibonacci(int n)
{
    if (n == 0)
    {
        return 0;
    }

    if (n == 1)
    {
        return 1;
    }

    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main()
{
    int n;
    int i;

    printf("Enter number of terms: ");
    scanf("%d", &n);

    printf("Fibonacci Series: ");

    for (i = 0; i < n; i++)
    {
        printf("%d ", fibonacci(i));
    }

    return 0;
}
Input: 
Enter number of terms: 7
Output: 
Fibonacci Series: 0 1 1 2 3 5 8

Program 3: Calculate the Sum of Natural Numbers Till 10 Using Recursion

#include <stdio.h>

int sum(int n)
{
    if (n == 0)
    {
        return 0;
    }

    return n + sum(n - 1);
}

int main()
{
    printf("Sum of natural numbers till 10 = %d", sum(10));

    return 0;
}

No input required.

Sum of natural numbers till 10 = 55

Program 4: Convert a Number into Binary Using Recursion

#include <stdio.h>

void binary(int n)
{
    if (n == 0)
    {
        return;
    }

    binary(n / 2);

    printf("%d", n % 2);
}

int main()
{
    int n;

    printf("Enter a decimal number: ");
    scanf("%d", &n);

    printf("Binary number = ");

    if (n == 0)
    {
        printf("0");
    }
    else
    {
        binary(n);
    }

    return 0;
}
Input: 
Enter a decimal number: 10
Output: 
Binary number = 1010

Program 5: Find the Largest Element in an Integer Array Using Recursion

#include <stdio.h>

int largest(int a[], int n)
{
    int max;

    if (n == 1)
    {
        return a[0];
    }

    max = largest(a, n - 1);

    if (a[n - 1] > max)
    {
        return a[n - 1];
    }

    return max;
}

int main()
{
    int a[100];
    int n;
    int i;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements: ");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Largest element = %d", largest(a, n));

    return 0;
}
Input: 
Enter array size: 5
Enter array elements: 12 45 23 67 34
Output: 
Largest element = 67

Program 7: Enter a Number from User and Add All Digits Using Recursion

#include <stdio.h>

int sumDigits(int n)
{
    if (n == 0)
    {
        return 0;
    }

    return (n % 10) + sumDigits(n / 10);
}

int main()
{

    printf("Enter a number: ");

    printf("Sum of digits = %d", sumDigits(n));

    return 0;

Program 8: Find Arithmetic Mean of an Integer Array Using Recursion

#include <stdio.h>

int arraySum(int a[], int n)
{
    if (n == 0)
    {
        return 0;
    }

    return a[n - 1] + arraySum(a, n - 1);
}

int main()
{
    int a[100];
    int n;
    int i;
    float mean;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements: ");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    mean = (float) arraySum(a, n) / n;

    printf("Arithmetic Mean = %.2f", mean);

    return 0;
}
Input: 
Enter array size: 5
Enter array elements: 10 20 30 40 50
Output: 
Arithmetic Mean = 30.00
