PROGRAM 1 - SINGLY LINKED LIST (INTEGER)

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;

int isEmpty() {
    return head == NULL;
}

int isFull() {
    struct Node *p = malloc(sizeof(struct Node));
    if (p == NULL)
        return 1;
    free(p);
    return 0;
}

void insertBegin(int x) {
    struct Node *p = malloc(sizeof(struct Node));
    if (p == NULL)
        return;
    p->data = x;
    p->next = head;
    head = p;
}

void insertEnd(int x) {
    struct Node *p = malloc(sizeof(struct Node));
    struct Node *t = head;
    if (p == NULL)
        return;
    p->data = x;
    p->next = NULL;
    if (head == NULL) {
        head = p;
        return;
    }
    while (t->next != NULL)
        t = t->next;
    t->next = p;
}

void insertPos(int x, int pos) {
    struct Node *p = malloc(sizeof(struct Node));
    struct Node *t = head;
    int i;
    if (p == NULL || pos < 1)
        return;
    p->data = x;
    if (pos == 1) {
        p->next = head;
        head = p;
        return;
    }
    for (i = 1; i < pos - 1 && t != NULL; i++)
        t = t->next;
    if (t == NULL) {
        free(p);
        return;
    }
    p->next = t->next;
    t->next = p;
}

void deleteBegin() {
    struct Node *t = head;
    if (head == NULL)
        return;
    head = head->next;
    free(t);
}

void deleteEnd() {
    struct Node *t = head;
    if (head == NULL)
        return;
    if (head->next == NULL) {
        free(head);
        head = NULL;
        return;
    }
    while (t->next->next != NULL)
        t = t->next;
    free(t->next);
    t->next = NULL;
}

void deletePos(int pos) {
    struct Node *t = head;
    struct Node *p;
    int i;
    if (head == NULL || pos < 1)
        return;
    if (pos == 1) {
        deleteBegin();
        return;
    }
    for (i = 1; i < pos - 1 && t != NULL; i++)
        t = t->next;
    if (t == NULL || t->next == NULL)
        return;
    p = t->next;
    t->next = p->next;
    free(p);
}

void display() {
    struct Node *t = head;
    while (t != NULL) {
        printf("%d ", t->data);
        t = t->next;
    }
    printf("\\n");
}

void search(int x) {
    struct Node *t = head;
    int pos = 1;
    while (t != NULL) {
        if (t->data == x) {
            printf("Found at position %d\\n", pos);
            return;
        }
        t = t->next;
        pos++;
    }
    printf("Not Found\\n");
}

int main() {
    insertBegin(20);
    insertBegin(10);
    insertEnd(40);
    insertPos(30, 3);
    display();
    search(30);
    deleteBegin();
    deleteEnd();
    deletePos(1);
    display();
    printf("Empty: %d\\n", isEmpty());
    printf("Full: %d\\n", isFull());
    return 0;
}

        n->next = t->next;
        t->coeff = c;
        t->power = p;
        t->next = n;
        return;

    n->next = t->next;
    t->next = n;
