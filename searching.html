Program 1: Linear Search Through Iteration on Integer Array

#include <stdio.h>

int main()
{
    int a[100], n, key, i, found = 0;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &key);

    for (i = 0; i < n; i++)
    {
        if (a[i] == key)
        {
            printf("Element found at position %d", i + 1);
            found = 1;
            break;
        }
    }

    if (found == 0)
    {
        printf("Element not found");
    }

    return 0;
}

Enter array size: 5
Enter array elements:
10 20 30 40 50
Enter value to search: 30
Output:

Program 2: Linear Search Through Recursion on String

#include <stdio.h>
#include <string.h>

int linearSearch(char str[], char key, int i)
{
    if (str[i] == '\0')
    {
        return -1;
    }

    if (str[i] == key)
    {
        return i;
    }

    return linearSearch(str, key, i + 1);
}

int main()
{
    char str[100], key;
    int result;

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Enter character to search: ");
    scanf(" %c", &key);

    result = linearSearch(str, key, 0);

    if (result == -1)
    {
        printf("Character not found");
    }
    else
    {
        printf("Character found at position %d", result + 1);
    }

    return 0;
}
Input: 
Enter a string: COMPUTER
Enter character to search: P
Output: 
Character found at position 4

Program 5: Binary Search Through Iteration on Integer Array

#include <stdio.h>

int main()
{
    int a[100], n, key, low, high, mid, i;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter sorted array elements:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;

    while (low <= high)
    {
        mid = (low + high) / 2;

        if (a[mid] == key)
        {
            printf("Element found at position %d", mid + 1);
            return 0;
        }
        else if (key < a[mid])
        {
            high = mid - 1;
        }
        else
        {
            low = mid + 1;
        }
    }

    printf("Element not found");

    return 0;
}

Program 6: Binary Search Through Recursion on String

#include <stdio.h>
#include <string.h>

int binarySearch(char str[], char key, int low, int high)
{
    int mid;

    if (low > high)
    {
        return -1;
    }

    mid = (low + high) / 2;

    if (str[mid] == key)
    {
        return mid;
    }

    if (key < str[mid])
    {
        return binarySearch(str, key, low, mid - 1);
    }

    return binarySearch(str, key, mid + 1, high);
}

int main()
{
    char str[100], key;
    int result, n;

    printf("Enter a sorted string: ");
    scanf("%s", str);

    n = strlen(str);

    printf("Enter character to search: ");
    scanf(" %c", &key);

    result = binarySearch(str, key, 0, n - 1);

    if (result == -1)
    {
        printf("Character not found");
    }
    else
    {
        printf("Character found at position %d", result + 1);
    }

    return 0;
}

Program 9: Calculate Best and Worst Complexity of Linear Search

#include <stdio.h>
#include <time.h>

int main()
{
    int a[100000], n, key, i;
    int comparisons;
    clock_t start, end;
    double timeTaken;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (i = 0; i < n; i++)
    {
        a[i] = i + 1;
    }

    printf("Enter value to search: ");
    scanf("%d", &key);

    comparisons = 0;

    start = clock();

    for (i = 0; i < n; i++)
    {
        comparisons++;

        if (a[i] == key)
        {
            break;
        }
    }

    end = clock();

    timeTaken = (double)(end - start) / CLOCKS_PER_SEC;

    if (i == n)
    {
        printf("Element not found\n");
    }
    else
    {
        printf("Element found at position %d\n", i + 1);
    }

    printf("Number of comparisons: %d\n", comparisons);
    printf("Time taken: %f seconds\n", timeTaken);

    printf("\nBest Case Complexity: O(1)");
    printf("\nWorst Case Complexity: O(n)");

    return 0;
}
Input: 
Enter array size: 5
Enter value to search: 1
Output: 
Element found at position 1
Number of comparisons: 1
Time taken: 0.000002 seconds

Best Case Complexity: O(1)
Worst Case Complexity: O(n)

Program 10: Calculate Best and Worst Complexity of Binary Search

#include <stdio.h>
#include <time.h>

int main()
{
    int a[100000], n, key;
    int low, high, mid, i, comparisons;
    clock_t start, end;
    double timeTaken;

    printf("Enter array size: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        a[i] = i + 1;
    }

    printf("Enter value to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;
    comparisons = 0;

    start = clock();

    while (low <= high)
    {
        mid = (low + high) / 2;
        comparisons++;

        if (a[mid] == key)
        {
            break;
        }
        else if (key < a[mid])
        {
            high = mid - 1;
        }
        else
        {
            low = mid + 1;
        }
    }

    end = clock();

    timeTaken = (double)(end - start) / CLOCKS_PER_SEC;

    if (low <= high)
    {
        printf("Element found at position %d\n", mid + 1);
    }
    else
    {
        printf("Element not found\n");
    }

    printf("Number of comparisons: %d\n", comparisons);
    printf("Time taken: %f seconds\n", timeTaken);

    printf("\nBest Case Complexity: O(1)");
    printf("\nWorst Case Complexity: O(log n)");

    return 0;
}

Enter array size: 5
Enter value to search: 3
Output: 
Element found at position 3
Number of comparisons: 1
Time taken: 0.000002 seconds

Best Case Complexity: O(1)
Worst Case Complexity: O(log n)
