Program 1: Construct Binary Tree from Preorder and Inorder Traversal

#include <stdio.h>
#include <stdlib.h>

int pre[100], in[100], p;

typedef struct N {
    int d;
    struct N *l, *r;
} N;

N *build(int a, int b) {
    if (a > b)
        return NULL;

    N *t = malloc(sizeof(N));
    t->d = pre[p++];

    int k = a;

    while (in[k] != t->d)
        k++;

    t->l = build(a, k - 1);
    t->r = build(k + 1, b);

    return t;
}

void post(N *t) {
    if (t) {
        post(t->l);
        post(t->r);
        printf("%d ", t->d);
    }
}

int main() {
    int n, i;

    scanf("%d", &n);

    for (i = 0; i < n; i++)
        scanf("%d", &pre[i]);

    for (i = 0; i < n; i++)
        scanf("%d", &in[i]);

    p = 0;

    post(build(0, n - 1));

    return 0;
}
Input: 
5
1 2 4 5 3
4 2 5 1 3
Output: 
4 5 2 3 1

Program 6: Find the Maximum Depth of Binary Tree

#include <stdio.h>
#include <stdlib.h>

    N *t = malloc(sizeof(N));

    return a > b ? a : b;

    r->l->r = node(5);

Program 7: Create Binary Search Tree Using Arrays

#include <stdio.h>

#define M 100

    int i = 1;

    while (i < M && t[i]) {
        if (x < t[i])
            i *= 2;
        else if (x > t[i])
            i = 2 * i + 1;
        else
            return;
    }

    if (i < M)
        t[i] = x;
}

void inorder(int i) {
    if (i < M && t[i]) {

        inorder(2 * i + 1);

Program 9: Find Inorder Predecessor and Inorder Successor of BST

#include <stdio.h>
#include <stdlib.h>

typedef struct N {
    int d;
    struct N *l, *r;
} N;

N *ins(N *t, int x) {
    if (!t) {
        t = malloc(sizeof(N));
        t->d = x;
        t->l = t->r = NULL;
    }
    else if (x < t->d)
        t->l = ins(t->l, x);
    else if (x > t->d)
        t->r = ins(t->r, x);

    return t;
}

int main() {
    int a[] = {50, 30, 70, 20, 40, 60, 80};
    int i, key = 50;

    N *t = NULL;
    N *p = NULL;
    N *s = NULL;
    N *q;

    for (i = 0; i < 7; i++)
        t = ins(t, a[i]);

    q = t;

    while (q && q->d != key) {
        if (key < q->d) {
            s = q;
            q = q->l;
        }
        else {
            p = q;
            q = q->r;
        }
    }

    if (q && q->l) {
        p = q->l;
        while (p->r)
            p = p->r;
    }

    if (q && q->r) {
        s = q->r;
        while (s->l)
            s = s->l;
    }

    printf("Predecessor: %d\nSuccessor: %d", p->d, s->d);

    return 0;
}

No input (BST and key are created in the program).
Output:

Successor: 60

Program 10: Check Whether Binary Search Tree Contains Dead End

#include <stdio.h>
#include <stdlib.h>

typedef struct N {
    int d;
    struct N *l, *r;
} N;

N *ins(N *t, int x) {
    if (!t) {
        t = malloc(sizeof(N));
        t->d = x;
        t->l = t->r = NULL;
    }
    else if (x < t->d)
        t->l = ins(t->l, x);
    else if (x > t->d)
        t->r = ins(t->r, x);

    return t;
}

int dead(N *t, int lo, int hi) {
    if (!t)
        return 0;

    if (lo == hi)
        return 1;

    return dead(t->l, lo, t->d - 1) ||
           dead(t->r, t->d + 1, hi);
}

int main() {
    int a[] = {8, 5, 9, 2, 7, 1};
    int i;
    N *r = NULL;

    for (i = 0; i < 6; i++)
        r = ins(r, a[i]);

    printf(dead(r, 1, 100) ? "Dead End Found" : "No Dead End");

    return 0;
}

No input (values are inserted in the program).
Output: 
Dead End Found

Program 14: Print Preorder Traversal to Postorder Traversal of BST

#include <stdio.h>

void post(int a[], int *i, int n, int lo, int hi) {
    if (*i == n || a[*i] < lo || a[*i] > hi)
        return;

    int x = a[(*i)++];

    post(a, i, n, lo, x - 1);
    post(a, i, n, x + 1, hi);

    printf("%d ", x);
}

int main() {
    int a[] = {40, 30, 35, 80, 100};
    int i = 0;

    post(a, &i, 5, -100000, 100000);

    return 0;
}

Program 17: Create Strictly Binary Tree from User Input Dynamically

#include <stdio.h>
#include <stdlib.h>

typedef struct N {
    int d;
    struct N *l, *r;
} N;

N *create() {
    int x;

    scanf("%d", &x);

    if (x == -1)
        return NULL;

    N *t = malloc(sizeof(N));

    t->d = x;
    t->l = create();
    t->r = create();

    return t;
}

int strict(N *t) {
    if (!t)
        return 1;

    if ((t->l == NULL) != (t->r == NULL))
        return 0;

    return strict(t->l) && strict(t->r);
}

int main() {
    N *r;

    printf("Enter preorder values (-1 for NULL): ");

    r = create();

    printf(strict(r) ? "Strictly Binary Tree" : "Not a Strictly Binary Tree");

    return 0;
}
